'use client';

import { useState, useEffect, useCallback, useRef } from 'react';
import PromptManager, { type PromptManagerHandle } from '@/components/prompt-manager';
import { useToast } from '@/hooks/use-toast';
import { getOptimizedPrompt, getSummarizedPrompt } from './actions';
import type { OptimizePromptOutput, SummarizePromptOutput } from './schemas';
import LogViewer from '@/components/log-viewer';
import { Button } from '@/components/ui/button';
import { BookText, Sparkles, Wand2, RefreshCw, Settings, BrainCircuit, Mic } from 'lucide-react';
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card';
import { Textarea } from '@/components/ui/textarea';
import { Skeleton } from '@/components/ui/skeleton';
import FinalPrompt from '@/components/final-prompt';
import { Label } from '@/components/ui/label';
import { Switch } from '@/components/ui/switch';
import { ScrollArea } from '@/components/ui/scroll-area';

export type Prompt = {
  id: string;
  name: string;
  text: string;
};

const initialPrompts: Prompt[] = [
  { id: '1', name: 'Summarize Text', text: 'Please summarize the following text in three key points: ' },
  { id: '2', name: 'Explain Like I\'m 5', text: 'Explain the following concept like I am 5 years old: ' },
  { id: '3', name: 'Translate to French', text: 'Translate the following to French, ensuring a natural and fluent tone: ' },
  { id: '4', name: 'Generate Tweet Ideas', text: 'Generate 5 tweet ideas based on this topic: ' },
];

export type Stats = {
    originalWordCount: number;
    finalWordCount: number;
    originalCharCount: number;
    finalCharCount: number;
    promptTokens: number | undefined;
    completionTokens: number | undefined;
} | null;


export default function Home() {
  const [prompts, setPrompts] = useState<Prompt[]>([]);
  const [injectionEnabled, setInjectionEnabled] = useState<boolean>(true);
  const [workspaceContent, setWorkspaceContent] = useState<string>('');
  
  const [finalContent, setFinalContent] = useState<string>('');
  const [stats, setStats] = useState<Stats>(null);
  const [isLoading, setIsLoading] = useState(false);

  const [isMounted, setIsMounted] = useState(false);
  const { toast } = useToast();
  
  const [provider, setProvider] = useState('openai');
  const [model, setModel] = useState('gpt-4o-mini');
  const [apiKey, setApiKey] = useState('');
  const [logs, setLogs] = useState<string[]>([]);
  const [isLogViewerOpen, setIsLogViewerOpen] = useState(false);

  const [isOptimizerEnabled, setIsOptimizerEnabled] = useState(true);
  const [isSummarizerEnabled, setIsSummarizerEnabled] = useState(true);

  const promptManagerRef = useRef<PromptManagerHandle>(null);

  useEffect(() => {
    setIsMounted(true);
  }, []);

  useEffect(() => {
    if (isMounted) {
      try {
        const storedPrompts = localStorage.getItem('prompt-injekt-prompts');
        if (storedPrompts) {
          const parsedPrompts = JSON.parse(storedPrompts);
          if (Array.isArray(parsedPrompts) && parsedPrompts.length > 0) {
            setPrompts(parsedPrompts);
          } else {
            setPrompts(initialPrompts);
          }
        } else {
          setPrompts(initialPrompts);
        }

        const storedProvider = localStorage.getItem('prompt-injekt-provider');
        if(storedProvider) setProvider(JSON.parse(storedProvider));

        const storedModel = localStorage.getItem('prompt-injekt-model');
        if(storedModel) setModel(JSON.parse(storedModel));

        const storedApiKey = localStorage.getItem('prompt-injekt-api-key');
        if(storedApiKey) setApiKey(JSON.parse(storedApiKey));

      } catch (error) {
        console.error("Failed to load settings from localStorage", error);
        setPrompts(initialPrompts);
      }
    }
  }, [isMounted]);

  useEffect(() => {
    if (isMounted) {
      if (prompts.length > 0) {
        try {
          localStorage.setItem('prompt-injekt-prompts', JSON.stringify(prompts));
        } catch (error) {
          console.error("Failed to save prompts to localStorage", error);
        }
      }
      try {
        localStorage.setItem('prompt-injekt-provider', JSON.stringify(provider));
        localStorage.setItem('prompt-injekt-model', JSON.stringify(model));
        localStorage.setItem('prompt-injekt-api-key', JSON.stringify(apiKey));
      } catch (error) {
        console.error("Failed to save settings to localStorage", error);
      }
    }
  }, [prompts, isMounted, provider, model, apiKey]);

  const handleInjectPrompt = useCallback((promptText: string) => {
    if (injectionEnabled) {
      setWorkspaceContent(prev => prev ? `${prev.trim()}\n\n${promptText}` : promptText);
      toast({
        title: "Prompt Injected",
        description: "The selected prompt has been added to the workspace.",
      });
    } else {
      toast({
        variant: "destructive",
        title: "Injection Disabled",
        description: "Enable the toggle to inject prompts.",
      });
    }
  }, [injectionEnabled, toast]);

  const handleCopyToClipboard = useCallback((promptText: string, promptName: string) => {
    navigator.clipboard.writeText(promptText).then(() => {
      toast({
        title: "Prompt Copied!",
        description: `"${promptName}" copied to clipboard.`,
      });
    }).catch(err => {
      console.error("Failed to copy text: ", err);
      toast({
        variant: "destructive",
        title: "Copy Failed",
        description: "Could not copy prompt to clipboard.",
      });
    });
  }, [toast]);

  const handleAddPrompt = (prompt: Omit<Prompt, 'id'>) => {
    const newPrompt = { ...prompt, id: new Date().toISOString() };
    setPrompts(prev => [newPrompt, ...prev]);
    toast({ title: "Prompt Added", description: `"${prompt.name}" has been saved.` });
  };

  const handleUpdatePrompt = (updatedPrompt: Prompt) => {
    setPrompts(prev => prev.map(p => p.id === updatedPrompt.id ? updatedPrompt : p));
    toast({ title: "Prompt Updated", description: `"${updatedPrompt.name}" has been updated.` });
  };

  const handleDeletePrompt = (promptId: string) => {
    const promptToDelete = prompts.find(p => p.id === promptId);
    setPrompts(prev => prev.filter(p => p.id !== promptId));
    if (promptToDelete) {
      toast({
        variant: "destructive",
        title: "Prompt Deleted",
        description: `"${promptToDelete.name}" has been removed.`,
      });
    }
  };

  const handleSaveSettings = (settings: { provider: string; model: string; apiKey: string }) => {
    setProvider(settings.provider);
    setModel(settings.model);
    setApiKey(settings.apiKey);
    toast({ title: "Settings Saved", description: "Your new settings have been saved." });
  };

  const handleGenerate = async (initialContent: string) => {
    if (initialContent.trim().length < 10) {
      toast({
          variant: 'destructive',
          title: 'Prompt too short',
          description: 'Please enter at least 10 characters to generate.'
      });
      return;
    }
    if (!isOptimizerEnabled && !isSummarizerEnabled) {
      toast({
          variant: 'destructive',
          title: 'No workflow enabled',
          description: 'Please enable at least one workflow step (Optimizer or Summarizer).'
      });
      return;
    }

    setIsLoading(true);
    setFinalContent('');
    setStats(null);
    setLogs([]);
    
    const settings = { provider, model, apiKey };
    let currentContent = initialContent;
    let intermediateContent = initialContent;
    let totalPromptTokens: number | undefined = 0;
    let totalCompletionTokens: number | undefined = 0;

    try {
      // 1. Optimizer Step
      if (isOptimizerEnabled) {
        const optimizeResult = await getOptimizedPrompt({ prompt: currentContent }, settings);
        if(optimizeResult.logs) setLogs(prev => [...prev, ...optimizeResult.logs!]);
        
        if (!optimizeResult.optimized_prompt) {
            throw new Error('Optimization failed. No suggestion was returned.');
        }
        currentContent = optimizeResult.optimized_prompt;
      }

      intermediateContent = currentContent;

      // 2. Summarizer Step
      if (isSummarizerEnabled) {
        const summarizeResult = await getSummarizedPrompt({ prompt: currentContent }, settings);
        if(summarizeResult.logs) setLogs(prev => [...prev, ...summarizeResult.logs!]);
        
        if (!summarizeResult.summarized_prompt) {
          throw new Error('Summarization failed. No summary was returned.');
        }

        if (summarizeResult.summarized_prompt.trim() === currentContent.trim()) {
            toast({
                title: "Already Concise",
                description: "The text could not be summarized further."
            });
        } else {
            currentContent = summarizeResult.summarized_prompt;
        }

        if (summarizeResult.usage?.prompt_tokens) totalPromptTokens = (totalPromptTokens || 0) + summarizeResult.usage.prompt_tokens;
        if (summarizeResult.usage?.completion_tokens) totalCompletionTokens = (totalCompletionTokens || 0) + summarizeResult.usage.completion_tokens;
      }

      // 3. Set Final State
      setFinalContent(currentContent);
      setStats({
          originalWordCount: initialContent.split(/\s+/).filter(Boolean).length,
          finalWordCount: currentContent.split(/\s+/).filter(Boolean).length,
          originalCharCount: initialContent.length,
          finalCharCount: currentContent.length,
          promptTokens: totalPromptTokens || undefined,
          completionTokens: totalCompletionTokens || undefined,
      });

      // Clear workspace only if the initial trigger was the main generate button
      if (initialContent === workspaceContent) {
          setWorkspaceContent('');
      }

    } catch (error: any) {
        toast({
            variant: 'destructive',
            title: 'Generation Failed',
            description: `${error.message} Please check logs for details.`
        });
    } finally {
        setIsLoading(false);
    }
  }


  const handleSaveFinalPrompt = (content: string) => {
    promptManagerRef.current?.openDialog(null, content);
  };

  const handleRegenerate = () => {
    if (finalContent) {
      handleGenerate(finalContent);
    }
  };

  const isGenerateDisabled = isLoading || !workspaceContent || (!isOptimizerEnabled && !isSummarizerEnabled);

  return (
    <div className="h-full flex flex-col">
      <header className="py-4 px-4 sm:px-8 border-b shrink-0">
        <div className="container mx-auto flex items-center justify-between">
          <div className="flex items-center gap-3">
            <div className="bg-primary p-2 rounded-lg">
              <svg
                xmlns="http://www.w3.org/2000/svg"
                width="24"
                height="24"
                viewBox="0 0 24 24"
                fill="none"
                stroke="hsl(var(--primary-foreground))"
                strokeWidth="2"
                strokeLinecap="round"
                strokeLinejoin="round"
                className="h-6 w-6"
              >
                <path d="m18 16 4-4-4-4" />
                <path d="m6 8-4 4 4 4" />
                <path d="m14.5 4-5 16" />
              </svg>
            </div>
            <h1 className="text-2xl font-bold font-headline text-primary-foreground select-none">
              Prompt Injekt
            </h1>
          </div>
          <LogViewer logs={logs} open={isLogViewerOpen} onOpenChange={setIsLogViewerOpen}>
            <Button variant="outline">
              <BookText className="mr-2 h-4 w-4" />
              View Logs
            </Button>
          </LogViewer>
        </div>
      </header>
      <main className="container mx-auto p-4 sm:p-8 flex-grow grid grid-cols-1 lg:grid-cols-3 gap-8 items-start min-h-0">
        <div className="lg:col-span-1 space-y-8 h-full flex flex-col">
          <PromptManager
            ref={promptManagerRef}
            prompts={prompts}
            injectionEnabled={injectionEnabled}
            onInjectionToggle={setInjectionEnabled}
            onInject={handleInjectPrompt}
            onAdd={handleAddPrompt}
            onUpdate={handleUpdatePrompt}
            onDelete={handleDeletePrompt}
            onCopy={(prompt) => handleCopyToClipboard(prompt.text, prompt.name)}
            settings={{ provider, model, apiKey }}
            onSaveSettings={handleSaveSettings}
          />
        </div>
        <div className="lg:col-span-2 h-full flex flex-col">
          <ScrollArea className="h-full">
            <div className="space-y-8 pr-6">
              <Card className="shadow-lg">
                  <CardHeader>
                      <CardTitle className='flex items-center gap-2'>
                          <Settings className="h-6 w-6 text-primary" />
                          Workflow Control
                      </CardTitle>
                      <CardDescription>
                      Enable or disable steps in the generation pipeline.
                      </CardDescription>
                  </CardHeader>
                  <CardContent className="flex items-center space-x-8">
                      <div className="flex items-center space-x-2">
                          <Switch id="optimizer-toggle" checked={isOptimizerEnabled} onCheckedChange={setIsOptimizerEnabled} />
                          <Label htmlFor="optimizer-toggle" className="flex items-center gap-2 text-base">
                              <BrainCircuit className="h-5 w-5" /> Optimizer
                          </Label>
                      </div>
                      <div className="flex items-center space-x-2">
                          <Switch id="summarizer-toggle" checked={isSummarizerEnabled} onCheckedChange={setIsSummarizerEnabled} />
                          <Label htmlFor="summarizer-toggle" className="flex items-center gap-2 text-base">
                              <Mic className="h-5 w-5" /> Summarizer
                          </Label>
                      </div>
                  </CardContent>
              </Card>

              <Card className="shadow-lg">
                  <CardHeader>
                      <CardTitle className='flex items-center gap-2'>
                          <Wand2 className="h-6 w-6 text-primary" />
                          Prompt Generation Workspace
                      </CardTitle>
                      <CardDescription>
                      Type, paste, or inject a prompt idea below. Click "Generate" to run it through the selected workflow.
                      </CardDescription>
                  </CardHeader>
                  <CardContent>
                      <Textarea
                      placeholder="e.g., 'Create a summary of a text...'"
                      value={workspaceContent}
                      onChange={(e) => setWorkspaceContent(e.target.value)}
                      className="min-h-96 text-base"
                      aria-label="Prompt idea text area"
                      />
                      <Button onClick={() => handleGenerate(workspaceContent)} disabled={isGenerateDisabled} className="mt-4 w-full">
                          <Sparkles className="mr-2 h-4 w-4" />
                          {isLoading ? 'Generating...' : 'Generate'}
                      </Button>
                  </CardContent>
              </Card>

              {isLoading ? (
                  <Card>
                      <CardHeader>
                        <CardTitle className='flex items-center gap-2'>
                          <Sparkles className="h-6 w-6 text-primary animate-pulse" />
                          Generating...
                        </CardTitle>
                      </CardHeader>
                      <CardContent className="space-y-4">
                          <Skeleton className="h-24 w-full" />
                          <Skeleton className="h-16 w-full" />
                      </CardContent>
                  </Card>
              ) : (
                  (finalContent || stats) && (
                      <FinalPrompt
                          content={finalContent}
                          stats={stats}
                          onSave={handleSaveFinalPrompt}
                          onRegenerate={handleRegenerate}
                          isRegenerateDisabled={isLoading}
                      />
                  )
              )}
            </div>
          </ScrollArea>
        </div>
      </main>
    </div>
  );
}

'use server';

import { optimizePrompt } from '@/ai/flows/optimize-prompt-flow';
import { summarizePrompt } from '@/ai/flows/summarize-prompt-flow';
import type { 
    OptimizePromptInputForClient, 
    OptimizePromptOutput,
    SummarizePromptInputForClient,
    SummarizePromptOutput
} from './schemas';


export async function getOptimizedPrompt(input: OptimizePromptInputForClient, settings: { apiKey: string; provider: string; model: string; }): Promise<OptimizePromptOutput> {
    const logs: string[] = [];
    try {
        if (!input.prompt || input.prompt.trim().length < 10) {
            logs.push("Prompt is too short for optimization. Returning empty.");
            return { optimized_prompt: "", logs };
        }

        if (!settings.apiKey) {
            logs.push("API Key is not set. Cannot optimize prompt.");
            return { optimized_prompt: "", logs };
        }

        const flowInput = {
            prompt: input.prompt,
            apiKey: settings.apiKey,
            provider: settings.provider,
            target_provider: "google" // As per original implementation
        }
        
        const result = await optimizePrompt(flowInput);
        return result;

    } catch (error: any) {
        logs.push(`Error fetching suggestions from Genkit flow (Optimize): ${error.message}`);
        return { optimized_prompt: "", logs };
    }
}


export async function getSummarizedPrompt(input: SummarizePromptInputForClient, settings: { apiKey: string; provider: string; model: string; }): Promise<SummarizePromptOutput> {
    const logs: string[] = [];
    try {
        if (!input.prompt || input.prompt.trim().length < 10) {
            logs.push("Prompt is too short for summarization. Returning empty.");
            return { summarized_prompt: "" };
        }

        if (!settings.apiKey) {
            logs.push("API Key is not set. Cannot summarize prompt.");
            return { summarized_prompt: "" };
        }
        
        const flowInput = {
            prompt: input.prompt,
            apiKey: settings.apiKey,
            provider: `${settings.provider}/${settings.model}`,
            output_sentences: 1
        }
        
        const result = await summarizePrompt(flowInput);
        return result;

    } catch (error: any) {
        logs.push(`Error fetching suggestions from Genkit flow (Summarize): ${error.message}`);
        return { summarized_prompt: "", logs };
    }
}

import { z } from 'zod';

// For Prompt Optimizer
export const OptimizePromptInputSchema = z.object({
  prompt: z.string().describe('The prompt text to be optimized.'),
});
export type OptimizePromptInputForClient = z.infer<typeof OptimizePromptInputSchema>;

export const OptimizePromptOutputSchema = z.object({
  optimized_prompt: z.string().describe('The optimized prompt returned by the API.'),
  logs: z.array(z.string()).optional().describe('An array of log messages from the flow execution.'),
});
export type OptimizePromptOutput = z.infer<typeof OptimizePromptOutputSchema>;

// For Prompt Summarizer
export const SummarizePromptInputSchema = z.object({
    prompt: z.string().describe('The prompt text to be summarized.'),
    apiKey: z.string().describe('The Eden AI API key.'),
    provider: z.string().describe('The source provider for summarization (e.g., "openai").'),
    output_sentences: z.number().describe('The number of sentences in the summary.'),
});
export type SummarizePromptInput = z.infer<typeof SummarizePromptInputSchema>;
export type SummarizePromptInputForClient = Omit<SummarizePromptInput, 'apiKey' | 'provider' | 'output_sentences'>;


const SummarizeUsageSchema = z.object({
    prompt_tokens: z.number().optional(),

'use server';
/**
 * @fileOverview Optimizes a user-provided prompt using the Eden AI API.
 *
 * - optimizePrompt - A function that takes a prompt and returns an optimized version.
 */

import { ai } from '@/ai/genkit';
import { z } from 'genkit';

const OptimizePromptInputSchema = z.object({
  prompt: z.string().describe('The prompt text to be optimized.'),
  apiKey: z.string().describe('The Eden AI API key.'),
  provider: z.string().describe('The source provider for optimization (e.g., "openai").'),
  target_provider: z.string().describe('The target provider for the optimized prompt (e.g., "google").'),
});
type OptimizePromptInput = z.infer<typeof OptimizePromptInputSchema>;

const OptimizePromptOutputSchema = z.object({
  optimized_prompt: z.string().describe('The optimized prompt returned by the API.'),
  logs: z.array(z.string()).optional().describe('An array of log messages from the flow execution.'),
});
type OptimizePromptOutput = z.infer<typeof OptimizePromptOutputSchema>;


export async function optimizePrompt(
  input: OptimizePromptInput
): Promise<OptimizePromptOutput> {
  return optimizePromptFlow(input);
}


const optimizePromptFlow = ai.defineFlow(
  {
    name: 'optimizePromptFlow',
    inputSchema: OptimizePromptInputSchema,
    outputSchema: OptimizePromptOutputSchema,
  },
  async (input) => {
    const logs: string[] = [];
    const url = "https://api.edenai.run/v2/text/prompt_optimization";
    
    const payload = {
      providers: input.provider,
      text: input.prompt,
      target_provider: input.target_provider,
      response_as_dict: true,
      attributes_as_list: false,
      show_original_response: false
    };

    logs.push(`Sending payload to Eden AI: ${JSON.stringify(payload, null, 2)}`);

    const headers = {
      "Authorization": `Bearer ${input.apiKey}`,
      "Content-Type": "application/json",
      "Accept": "application/json",
    };

    try {
      const response = await fetch(url, {
        method: 'POST',
        headers: headers,
        body: JSON.stringify(payload),
      });

      if (!response.ok) {
        const errorBody = await response.text();
        logs.push(`Eden AI API Error Response: ${response.status} ${errorBody}`);
        throw new Error(`API request failed with status ${response.status}`);
      }
      
      const result = await response.json();
      logs.push(`Received response from Eden AI: ${JSON.stringify(result, null, 2)}`);
      
      const providerResult = result[input.provider];

      if (providerResult && providerResult.items && providerResult.items.length > 0 && providerResult.items[0].text) {
        return { optimized_prompt: providerResult.items[0].text, logs };
      } else {
        logs.push(`Unexpected API response structure: ${JSON.stringify(result)}`);
        throw new Error("Could not extract optimized prompt from API response.");
      }

    } catch (error: any) {
      logs.push(`Error calling Eden AI API: ${error.message}`);
      return { optimized_prompt: "", logs };
    }
  }
);

'use client';

import React from 'react';
import { Card, CardContent, CardHeader, CardTitle } from './ui/card';
import { TrendingDown } from 'lucide-react';

interface SummaryStatsProps {
  originalWordCount: number;
  finalWordCount: number;
  originalCharCount: number;
  finalCharCount: number;
  promptTokens: number | undefined;
  completionTokens: number | undefined;
}

const StatCard = ({ title, value, footer }: { title: string; value: string; footer: string }) => (
    <div className="bg-background p-4 rounded-lg shadow">
      <p className="text-sm text-muted-foreground">{title}</p>
      <p className="text-2xl font-bold">{value}</p>
      <p className="text-xs text-muted-foreground">{footer}</p>
    </div>
);

export const SummaryStats: React.FC<SummaryStatsProps> = ({
  originalWordCount,
  finalWordCount,
  originalCharCount,
  finalCharCount,
  promptTokens,
  completionTokens
}) => {
  const calculateReduction = (original: number, final: number) => {
    if (original === 0 || original <= final) return 0;
    return Math.round(((original - final) / original) * 100);
  };

  const wordReduction = calculateReduction(originalWordCount, finalWordCount);
  const charReduction = calculateReduction(originalCharCount, finalCharCount);
  const tokenReduction = (promptTokens && completionTokens) ? calculateReduction(promptTokens, completionTokens) : null;

  return (
    <Card className="mt-6">
      <CardHeader>
        <CardTitle className="text-lg flex items-center gap-2">
            <TrendingDown className="h-5 w-5 text-accent" />
            Reduction Analysis
        </CardTitle>
      </CardHeader>
      <CardContent className="grid grid-cols-1 md:grid-cols-3 gap-4">
        <StatCard 
            title="Words"
            value={`${wordReduction}%`}
            footer={`${originalWordCount} → ${finalWordCount}`}
        />
        <StatCard 
            title="Characters"
            value={`${charReduction}%`}
            footer={`${originalCharCount} → ${finalCharCount}`}
        />
        {tokenReduction !== null && promptTokens !== undefined && completionTokens !== undefined ? (
            <StatCard 
                title="Tokens"
                value={`${tokenReduction}%`}
                footer={`${promptTokens} → ${completionTokens}`}
            />
        ) : (
            <div className="bg-background p-4 rounded-lg shadow flex flex-col justify-center items-center">
                <p className="text-sm text-muted-foreground">Token data not available.</p>
            </div>
        )}
      </CardContent>
    </Card>
  );
};
